(defun takt/share-window (win mode)
  "Attach the Takt header to WIN and set interaction MODE."
  (interactive)
  (let ((bar (make-overlay-bar win)))
    (setf (bar-mode bar) mode)
    (pcase mode
      ('view    (takt/lock-input bar))
      ('control (takt/grant-pointer bar))
      ('draw    (takt/enable-annotate bar)))
    bar))
;; sharing to 3 viewers — Robert (owner)
